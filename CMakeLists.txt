cmake_minimum_required(VERSION 3.21)
project(f2bgl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
pkg_check_modules(WILDMIDI QUIET IMPORTED_TARGET wildmidi)
pkg_check_modules(FLUIDSYNTH REQUIRED IMPORTED_TARGET fluidsynth)
find_package(ZLIB REQUIRED)
find_package(OpenGL REQUIRED)
if(NOT TARGET PkgConfig::WILDMIDI)
  find_path(WILDMIDI_INCLUDE_DIR wildmidi_lib.h)
  find_library(WILDMIDI_LIBRARY NAMES WildMidi wildmidi WildMidi.dll)
  if(NOT WILDMIDI_INCLUDE_DIR OR NOT WILDMIDI_LIBRARY)
    message(FATAL_ERROR "WildMIDI development files are required. Set WILDMIDI_INCLUDE_DIR and WILDMIDI_LIBRARY, or provide a wildmidi pkg-config file.")
  endif()
  add_library(WildMidi::WildMidi UNKNOWN IMPORTED)
  set_target_properties(WildMidi::WildMidi PROPERTIES
    IMPORTED_LOCATION "${WILDMIDI_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${WILDMIDI_INCLUDE_DIR}")
  set(F2BGL_WILDMIDI_TARGET WildMidi::WildMidi)
else()
  set(F2BGL_WILDMIDI_TARGET PkgConfig::WILDMIDI)
endif()
set(F2BGL_SOURCES cabinet.cpp camera.cpp collision.cpp cutscene.cpp cutscenecin.cpp cutscenedps.cpp decoder.cpp file.cpp font.cpp game.cpp icons.cpp input.cpp installer.cpp inventory.cpp main.cpp mdec.cpp menu.cpp mixer.cpp opcodes.cpp raycast.cpp render.cpp resource.cpp saveload.cpp scaler.cpp screenshot.cpp sound.cpp spritecache.cpp stub.cpp texturecache.cpp trigo.cpp util.cpp xmiplayer.cpp)
add_executable(f2bgl ${F2BGL_SOURCES})
target_compile_definitions(f2bgl PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(f2bgl PRIVATE PkgConfig::SDL3 ${F2BGL_WILDMIDI_TARGET} PkgConfig::FLUIDSYNTH ZLIB::ZLIB OpenGL::GL)
if(MSVC)
  target_compile_options(f2bgl PRIVATE /W3 /wd4244 /wd4267)
else()
  target_compile_options(f2bgl PRIVATE -Wall -Wno-sign-compare -Wpedantic)
endif()
